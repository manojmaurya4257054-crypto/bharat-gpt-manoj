const chatDiv=document.getElementById('chat'); const inp=document.getElementById('inp'); async function send(){ const text=inp.value.trim(); if(!text) return; addMsg(text,'user'); inp.value=''; const tid=addMsg('Soch raha hu...','bot'); try{ const res=await fetch('https://text.pollinations.ai/openai',{ method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ model:'openai', messages:[{role:'system',content:'Tum Bharat GPT ho, Manoj ne banaya hai. Hindi me short, helpful jawab do.'},{role:'user',content:text}], stream:false }) }); const data=await res.json(); document.getElementById(tid).remove(); let ans=data.choices[0].message.content; addMsg(ans,'bot'); }catch(e){ document.getElementById(tid).remove(); addMsg('Thoda wait karke fir try karo bhai, server busy hai','bot'); } } function addMsg(t,c){ const id='m'+Date.now(); const div=document.createElement('div'); div.className='msg '+c; div.id=id; div.innerText=t; chatDiv.appendChild(div); chatDiv.scrollTop=chatDiv.scrollHeight; return id; }
